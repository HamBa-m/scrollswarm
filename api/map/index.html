    Map & Entities API

    The map and items modules govern the scrolling airspace corridor, procedural multi-scale obstacle generation, target waypoints, and the guaranteed navigable passage invariant (\(G_{\min} \ge 110\text{ px}\)).

    Overview

    • Airspace Geometry: Configurable \(W \times H\) 2D corridor with ceiling and floor safety margins.
    • Scrolling Motion: Global horizontal leftward drift (\(v_{\text{scroll}} = 1\text{ px/step}\)).
    • Procedural Obstacles: Ground structures, overhead barriers, and oscillating mid-air debris.
    • Reconnaissance Targets: Stochastic waypoints with proximity-dependent camera Bernoulli trials.

    Quick Usage

    from scrollswarm.map import ScrollingMap, MapConfig

config = MapConfig(
    width=800,
    height=400,
    scroll_speed=1,
    min_corridor_gap=110,
    seed=42,
)

scrolling_map = ScrollingMap(config)
scrolling_map.reset(seed=42)

# Step map world
scrolling_map.step()
print(f"Active obstacles: {len(scrolling_map.obstacles)}, Targets: {len(scrolling_map.targets)}")

    class ScrollingMap:
    """
    Manages the scrolling aerial corridor, multi-scale procedural obstacles,
    guaranteed navigable passage, and stochastic inspection targets.

    The map advances one pixel per step (``scroll_speed = 1``) in the leftward
    direction.  Obstacles are procedurally generated off-screen to the right
    and culled when they scroll past the left edge.  The minimum navigable gap
    ``G_min >= min_corridor_gap`` is enforced at spawn time.

    Attributes:
        config: ``MapConfig`` instance governing all corridor parameters.
        width: Corridor width in pixels.
        height: Corridor height in pixels.
        ceiling_y: Upper navigable boundary ``y`` in pixels.
        floor_y: Lower navigable boundary ``y`` in pixels.
        obstacles: List of active ``Obstacle`` polygon entities.
        targets: List of active ``Target`` waypoint entities.
        rng: ``numpy.random.Generator`` used for all stochastic decisions.
        global_step: Monotonically increasing step counter.
        top_heights: Per-column ceiling boundary array (shape ``(W,)``).
        bot_heights: Per-column floor inset array (shape ``(W,)``).
    """

    def __init__(self, config: Optional[MapConfig] = None):
        """
        Initializes the ScrollingMap.

        Args:
            config: ``MapConfig`` dataclass instance.  Defaults to
                ``MapConfig()`` (800×400 corridor, standard spawn rates).
        """
        self.config = config or MapConfig()
        self.width = self.config.width
        self.height = self.config.height
        self.rng = np.random.default_rng(self.config.seed)

        self.min_corridor_gap = self.config.min_corridor_gap
        self.ceiling_y = self.config.ceiling_y
        self.floor_y = self.height - self.config.floor_margin
        self.global_step: int = 0

        # Boundary arrays (for backward compatibility with column-based checks)
        self.top_heights = np.full(self.width, self.ceiling_y, dtype=np.float32)
        self.bot_heights = np.full(
            self.width, self.config.floor_margin, dtype=np.float32
        )

        # Active dynamic entities
        self.obstacles: List[Obstacle] = []
        self.targets: List[Target] = []

        # Spawning timers
        self._steps_since_obstacle = 40
        self._steps_since_target = 30

        # Cached contiguous geometry buffers for current step
        self._geo_cache: Optional[
            Tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray]
        ] = None

        if getattr(self.config, "preseed", True):
            self._preseed_corridor()

    def reset(self, seed: Optional[int] = None) -> None:
        """Resets the map state, entities, and random number generator."""
        if seed is not None:
            self.rng = np.random.default_rng(seed)
        self.global_step = 0
        self._steps_since_obstacle = 40
        self._steps_since_target = 30
        self.obstacles.clear()
        self.targets.clear()
        self._geo_cache = None
        if getattr(self.config, "preseed", True):
            self._preseed_corridor()

    def _create_ground_structure(self, spawn_x: float, max_h: float) -> Obstacle:
        """Creates a ground-standing building or tower."""
        width = float(self.rng.uniform(30.0, 70.0))
        h = float(self.rng.uniform(40.0, max(50.0, max_h)))
        center_y = float(self.floor_y - h / 2.0)
        return Obstacle(
            x=float(spawn_x),
            y=center_y,
            base_y=center_y,
            width=width,
            height=h,
            obstacle_type=ObstacleType.STRUCTURE,
            vx=1.0,
        )

    def _create_overhead_structure(self, spawn_x: float, max_depth: float) -> Obstacle:
        """Creates an overhead suspended structure extending down from the ceiling."""
        width = float(self.rng.uniform(30.0, 70.0))
        depth = float(self.rng.uniform(30.0, max(40.0, max_depth)))
        center_y = float(self.ceiling_y + depth / 2.0)
        return Obstacle(
            x=float(spawn_x),
            y=center_y,
            base_y=center_y,
            width=width,
            height=depth,
            obstacle_type=ObstacleType.OVERHEAD,
            vx=1.0,
        )

    def _create_floating_hazard(self, spawn_x: float) -> Obstacle:
        """Creates a mid-air polygonal floating hazard with subtle epicyclic oscillation."""
        radius = float(self.rng.uniform(14.0, 32.0))
        vy_amp = float(self.rng.uniform(4.0, 16.0))
        omega = float(self.rng.uniform(0.02, 0.06))
        phase = float(self.rng.uniform(0.0, 2 * np.pi))

        effective_h = radius + vy_amp
        y_min = self.ceiling_y + effective_h + 15.0
        y_max = self.floor_y - effective_h - 15.0

        if y_min >= y_max:
            base_y = float(self.height / 2.0)
            vy_amp = 0.0
        else:
            base_y = float(self.rng.uniform(y_min, y_max))

        return Obstacle(
            x=float(spawn_x),
            y=base_y,
            base_y=base_y,
            radius=radius,
            width=radius * 2.0,
            height=radius * 2.0,
            obstacle_type=ObstacleType.FLOATING,
            vx=1.0,
            vy_amplitude=vy_amp,
            omega=omega,
            phase=phase,
        )

    def _create_procedural_obstacle(self, spawn_x: float) -> Obstacle:
        """
        Samples an obstacle type (ground structure, overhead structure, or floating hazard)
        while strictly preserving the guaranteed navigable corridor gap (>= 110 px).
        """
        max_structure_h = (self.floor_y - self.ceiling_y) - self.min_corridor_gap - 10.0
        max_structure_h = max(35.0, max_structure_h)

        obs_kind = self.rng.choice(
            [ObstacleType.STRUCTURE, ObstacleType.OVERHEAD, ObstacleType.FLOATING],
            p=[0.45, 0.35, 0.20],
        )

        if obs_kind == ObstacleType.STRUCTURE:
            return self._create_ground_structure(spawn_x, max_structure_h)
        elif obs_kind == ObstacleType.OVERHEAD:
            return self._create_overhead_structure(spawn_x, max_structure_h)
        else:
            return self._create_floating_hazard(spawn_x)

    def _preseed_corridor(self) -> None:
        """Populates the initial corridor with starting obstacles and inspection targets."""
        if self.width < 250 or self.height < 120:
            return

        x_positions = [self.width * 0.35, self.width * 0.60, self.width * 0.82]
        for x in x_positions:
            self.obstacles.append(self._create_procedural_obstacle(x))

        target_positions = [self.width * 0.45, self.width * 0.70, self.width * 0.92]
        for x in target_positions:
            low_y = min(self.ceiling_y + 35.0, self.height / 2.0 - 10.0)
            high_y = max(self.floor_y - 35.0, self.height / 2.0 + 10.0)
            if high_y <= low_y:
                high_y = low_y + 10.0
            y = float(self.rng.uniform(low_y, high_y))
            self.targets.append(Target(x=x, y=y))

    def step(self) -> None:
        """Advances the simulation by 1 step: updates obstacles, targets, and handles spawning."""
        self.global_step += 1
        self._steps_since_obstacle += 1
        self._steps_since_target += 1

        # 1. Advance dynamic obstacles
        for obs in self.obstacles:
            obs.step()

        # 2. Advance targets
        for tgt in self.targets:
            tgt.step()

        # 3. Cull offscreen entities
        self.obstacles = [obs for obs in self.obstacles if not obs.is_offscreen]
        self.targets = [tgt for tgt in self.targets if not tgt.is_offscreen]

        # 4. Stochastic spawning at right edge
        if self._steps_since_obstacle >= self.config.obstacle_spawn_interval:
            if self.rng.random() < 0.40:
                spawn_x = float(self.width + self.rng.uniform(30.0, 70.0))
                self.obstacles.append(self._create_procedural_obstacle(spawn_x))
                self._steps_since_obstacle = 0

        if self._steps_since_target >= self.config.target_spawn_interval:
            if self.rng.random() < 0.35:
                spawn_x = float(self.width + self.rng.uniform(20.0, 50.0))
                low_y = min(self.ceiling_y + 35.0, self.height / 2.0 - 10.0)
                high_y = max(self.floor_y - 35.0, self.height / 2.0 + 10.0)
                if high_y <= low_y:
                    high_y = low_y + 10.0
                y = float(self.rng.uniform(low_y, high_y))
                self.targets.append(Target(x=spawn_x, y=y))
                self._steps_since_target = 0

        # Invalidate geometry buffer cache on step
        self._geo_cache = None

    def get_geometry_buffers(
        self,
    ) -> Tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray]:
        """Returns cached contiguous NumPy geometry buffers for current map step."""
        if self._geo_cache is None:
            from scrollswarm.geometry import extract_geometry_buffers

            self._geo_cache = extract_geometry_buffers(self.obstacles, self.targets)
        return self._geo_cache

    def update(self) -> None:
        """Alias for step()."""
        self.step()

class Obstacle:
    """
    A multi-scale 2D obstacle (ground structure, overhead barrier, or floating hazard).
    Modeled as an arbitrary convex polygon with drift velocity and optional oscillation:
        x(t) = x_0 - v_x * t
        y(t) = y_base + A * sin(omega * t + phase)
    """

    x: float
    y: float
    base_y: float = 0.0
    radius: float = 20.0
    width: float = 40.0
    height: float = 60.0
    obstacle_type: ObstacleType = ObstacleType.STRUCTURE
    vertices_rel: List[Tuple[float, float]] = field(default_factory=list)
    vx: float = 1.0
    vy_amplitude: float = 0.0
    omega: float = 0.0
    phase: float = 0.0
    step_count: int = 0

    def __post_init__(self):
        if self.base_y == 0.0:
            self.base_y = self.y

        if not self.vertices_rel:
            if (
                self.obstacle_type == ObstacleType.STRUCTURE
                or self.obstacle_type == ObstacleType.OVERHEAD
            ):
                # Rectangular structure
                hw = self.width / 2.0
                hh = self.height / 2.0
                self.vertices_rel = [
                    (-hw, -hh),
                    (hw, -hh),
                    (hw, hh),
                    (-hw, hh),
                ]
                self.radius = float(np.sqrt(hw * hw + hh * hh))
            else:
                # Regular / irregular polygon for floating debris
                angles = np.linspace(0, 2 * np.pi, 7)[:-1]
                self.vertices_rel = [
                    (float(self.radius * np.cos(a)), float(self.radius * np.sin(a)))
                    for a in angles
                ]

    def step(self) -> None:
        """Advances the obstacle's horizontal drift and vertical oscillation."""
        self.x -= self.vx
        self.step_count += 1
        if self.vy_amplitude > 0.0:
            self.y = self.base_y + self.vy_amplitude * np.sin(
                self.omega * self.step_count + self.phase
            )

    @property
    def vertices(self) -> List[Tuple[float, float]]:
        """Returns the absolute 2D coordinates of the polygon's vertices."""
        return [(self.x + dx, self.y + dy) for dx, dy in self.vertices_rel]

    @property
    def edges(self) -> List[Tuple[Tuple[float, float], Tuple[float, float]]]:
        """Returns the consecutive edge line segments [(v0, v1), (v1, v2), ...]."""
        verts = self.vertices
        n = len(verts)
        return [(verts[i], verts[(i + 1) % n]) for i in range(n)]

    @property
    def is_offscreen(self) -> bool:
        """Returns True if the entire obstacle has scrolled past the left viewport edge."""
        if self.vertices_rel:
            max_rel_x = max(dx for dx, _ in self.vertices_rel)
            return self.x + max_rel_x < 0.0
        return self.x + self.radius < 0.0

    def contains_point(self, px: float, py: float) -> bool:
        """Determines if point (px, py) lies inside the polygon using ray-casting."""
        verts = self.vertices
        n = len(verts)
        if n == 0:
            return False
        inside = False
        p1x, p1y = verts[0]
        for i in range(n + 1):
            p2x, p2y = verts[i % n]
            if py > min(p1y, p2y):
                if py <= max(p1y, p2y):
                    if px <= max(p1x, p2x):
                        if p1y != p2y:
                            xinters = (py - p1y) * (p2x - p1x) / (p2y - p1y) + p1x
                        if p1x == p2x or px <= xinters:
                            inside = not inside
            p1x, p1y = p2x, p2y
        return inside

    def distance_to_point(self, px: float, py: float) -> float:
        """Computes minimum Euclidean distance from (px, py) to the polygon boundary."""
        if self.contains_point(px, py):
            return 0.0

        if not self.edges:
            dx = px - self.x
            dy = py - self.y
            return float(max(0.0, np.sqrt(dx * dx + dy * dy) - self.radius))

        min_dist_sq = float("inf")
        for (ax, ay), (bx, by) in self.edges:
            vx = bx - ax
            vy = by - ay
            wx = px - ax
            wy = py - ay
            c1 = wx * vx + wy * vy
            if c1 <= 0.0:
                dist_sq = (px - ax) ** 2 + (py - ay) ** 2
            else:
                c2 = vx * vx + vy * vy
                if c2 <= c1:
                    dist_sq = (px - bx) ** 2 + (py - by) ** 2
                else:
                    b = c1 / c2
                    proj_x = ax + b * vx
                    proj_y = ay + b * vy
                    dist_sq = (px - proj_x) ** 2 + (py - proj_y) ** 2
            if dist_sq < min_dist_sq:
                min_dist_sq = dist_sq

        return float(np.sqrt(min_dist_sq))

class Target:
    """
    An inspection target entity requiring aerial photography within an inspection radius.
    Action uncertainty is modeled via stochastic capture probability:
        P_j = (1 - d_j / D_photo) * sensor_reliability
        P_team = 1 - Prod_j (1 - P_j)
    """

    x: float
    y: float
    size: int = 8
    inspection_radius: float = 45.0
    sensor_reliability: float = 0.85
    collected: bool = False
    vx: float = 1.0

    def step(self) -> None:
        """Drifts with the base aerial corridor scroll."""
        self.x -= self.vx

    @property
    def is_offscreen(self) -> bool:
        """Returns True if the target has scrolled off the left edge."""
        return self.x + self.size < 0.0

    def sample_inspection(
        self,
        drones: List["Agent"],
        rng: np.random.Generator,
    ) -> bool:
        """
        Evaluates stochastic photo acquisition by nearby living drones within inspection radius.
        Returns True if a clear photograph is successfully captured.
        """
        if self.collected:
            return False

        # Gather nearby living drones
        inspecting_drones = []
        for drone in drones:
            if not drone.alive:
                continue
            dx = drone.x - self.x
            dy = drone.y - self.y
            dist = np.sqrt(dx * dx + dy * dy)
            if dist <= self.inspection_radius:
                inspecting_drones.append(dist)

        if not inspecting_drones:
            return False

        # Probability of team success under action uncertainty
        p_fail_team = 1.0
        for dist in inspecting_drones:
            # Proximity factor: closer drones have higher capture probability
            p_drone = (1.0 - (dist / self.inspection_radius)) * self.sensor_reliability
            p_drone = float(np.clip(p_drone, 0.05, 0.95))
            p_fail_team *= 1.0 - p_drone

        p_success_team = 1.0 - p_fail_team
        if rng.random() < p_success_team:
            self.collected = True
            return True

        return False
